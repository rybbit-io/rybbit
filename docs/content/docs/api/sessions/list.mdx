```
GET /api/sites/:site/sessions
```

Returns a paginated list of sessions with their analytics data.

### Path Parameters

<TypeTable
  type={{
    site: {
      description: 'Site ID or site identifier',
      type: 'string | number',
      required: true,
    },
  }}
/>

### Query Parameters

Accepts all [Common Parameters](/docs/api/getting-started#common-parameters) plus the following. An invalid `view`, `sort_by`, `sort_order`, `include_goals`, or min/max value returns `400`.

<TypeTable
  type={{
    page: {
      description: 'Page number (1-indexed). Invalid values fall back to 1.',
      type: 'number',
      default: 1,
    },
    limit: {
      description: 'Number of results per page, capped at 10,000. Invalid values fall back to 100.',
      type: 'number',
      default: 100,
    },
    user_id: {
      description: 'Only sessions of this user. Matches the identified user ID, or the device ID of anonymous sessions.',
      type: 'string',
      required: false,
    },
    session_id: {
      description: 'Only this session',
      type: 'string',
      required: false,
    },
    identified_only: {
      description: 'Only return sessions from identified users',
      type: 'string',
      typeDescription: '"true" | "false"',
      default: 'false',
    },
    view: {
      description: 'Preset view. identified: identified users. replay: sessions with a replay. converted: completed at least one goal (turns on include_goals); API keys without the goals:read scope get no sessions for this view. bounced: exactly one pageview. errors: at least one error.',
      type: 'string',
      typeDescription: '"all" | "identified" | "replay" | "converted" | "bounced" | "errors"',
      default: 'all',
    },
    sort_by: {
      description: 'Sort column. events sorts by custom events plus autocaptured interactions. Ties fall back to session start, then session ID.',
      type: 'string',
      typeDescription: '"started" | "ended" | "duration" | "pageviews" | "events" | "errors"',
      default: 'ended',
    },
    sort_order: {
      description: 'Sort direction',
      type: 'string',
      typeDescription: '"asc" | "desc"',
      default: 'desc',
    },
    include_goals: {
      description: "Add converted_goals to each session. Only the site's 50 oldest goals are matched. API keys without the goals:read scope get an empty list.",
      type: 'string',
      typeDescription: '"true" | "false"',
      default: 'false',
    },
    min_pageviews: {
      description: 'Minimum pageviews (inclusive)',
      type: 'number',
      required: false,
    },
    max_pageviews: {
      description: 'Maximum pageviews (inclusive)',
      type: 'number',
      required: false,
    },
    min_events: {
      description: 'Minimum events (inclusive). Counts custom events plus button clicks, copies, form submits, and input changes.',
      type: 'number',
      required: false,
    },
    max_events: {
      description: 'Maximum events (inclusive), counted like min_events',
      type: 'number',
      required: false,
    },
    min_duration: {
      description: 'Minimum session duration in seconds (inclusive)',
      type: 'number',
      required: false,
    },
    max_duration: {
      description: 'Maximum session duration in seconds (inclusive)',
      type: 'number',
      required: false,
    },
  }}
/>

<TypeTable

#### Session Object
